Create:


Code:

```tcl
namespace eval design {

    variable cells [dict create]

    proc add_cell {name type area} {

        variable cells

        dict set cells $name type $type
        dict set cells $name area $area
    }

    proc get_cells {} {

        variable cells

        return [dict keys $cells]
    }

    proc get_type {name} {

        variable cells

        return [dict get $cells $name type]
    }

    proc get_area {name} {

        variable cells

        return [dict get $cells $name area]
    }
}
```

# 13. report_parser.tcl

This is the most important file.

It will read the area report and extract:

```text
Cell
Area
```

Create:

```text
report_parser.tcl
```

Code:

```tcl
namespace eval parser {

    proc parse_area_report {filename} {

        if {![file exists $filename]} {
            error "Area report not found: $filename"
        }

        set fp [open $filename r]

        set count 0

        while {[gets $fp line] >= 0} {

            if {[regexp {
                Cell:\s+(\S+)\s+Area:\s+([0-9]+(\.[0-9]+)?)
            } $line -> cell area decimal]} {

                incr count

                set instance "U$count"

                design::add_cell $instance $cell $area
            }
        }

        close $fp

        return $count
    }
}
```

# 15. timing.tcl

Create:

```text
timing.tcl
```

Code:

```tcl
namespace eval timing {

    variable data [dict create]

    proc parse_report {filename} {

        if {![file exists $filename]} {
            error "Timing report not found: $filename"
        }

        variable data

        set fp [open $filename r]

        while {[gets $fp line] >= 0} {

            if {[regexp {
                WNS\s*=\s*([-+]?[0-9]*\.?[0-9]+)
            } $line -> value]} {

                dict set data WNS $value
            }

            if {[regexp {
                TNS\s*=\s*([-+]?[0-9]*\.?[0-9]+)
            } $line -> value]} {

                dict set data TNS $value
            }

            if {[regexp {
                Clock\s+Period\s*=\s*([-+]?[0-9]*\.?[0-9]+)
            } $line -> value]} {

                dict set data Clock_Period $value
            }
        }

        close $fp
    }

    proc get {key} {

        variable data

        if {[dict exists $data $key]} {
            return [dict get $data $key]
        }

        return ""
    }
}
```

# 16. area.tcl

Create:

```text
area.tcl
```

Code:

```tcl
namespace eval area {

    proc total_area {} {

        set total 0

        foreach cell [design::get_cells] {

            set cell_area [design::get_area $cell]

            set total [expr {$total + $cell_area}]
        }

        return $total
    }

    proc count_type {pattern} {

        set count 0

        foreach cell [design::get_cells] {

            set type [design::get_type $cell]

            if {[string match $pattern $type]} {
                incr count
            }
        }

        return $count
    }

    proc largest_cell {} {

        set largest ""
        set largest_area 0

        foreach cell [design::get_cells] {

            set cell_area [design::get_area $cell]

            if {$cell_area > $largest_area} {

                set largest_area $cell_area
                set largest $cell
            }
        }

        return $largest
    }
}
```

```tcl
# ========================================

source config.tcl

source timing.tcl
source area.tcl

# Start Flow
# ========================================

puts ""
puts "========================================"
puts "       VLSI QoR AUTOMATION"
puts "========================================"
puts ""


# ========================================
# Parse Area Report
# ========================================

puts "Reading area report..."

if {[catch {
    set cell_count [parser::parse_area_report $AREA_REPORT]
} result]} {

    puts "ERROR: $result"
    exit 1
}

puts "Area report processed."
puts "Cell count = $cell_count"


# ========================================
# Parse Timing Report
# ========================================

puts ""
puts "Reading timing report..."

if {[catch {
    timing::parse_report $TIMING_REPORT
} result]} {

    puts "ERROR: $result"
    exit 1
}

puts "Timing report processed."


# ========================================
# Calculate QoR
# ========================================

set total_area [area::total_area]

set nand_count [area::count_type "NAND*"]

set largest_cell [area::largest_cell]

set largest_area [design::get_area $largest_cell]

set wns [timing::get WNS]

set tns [timing::get TNS]

set period [timing::get Clock_Period]


# ========================================
# Check Limits
# ========================================

set area_status [vlsi::check_limit $total_area $AREA_LIMIT]

set wns_status [vlsi::check_non_negative $wns]

set tns_status [vlsi::check_non_negative $tns]


# ========================================
# Overall Status
# ========================================

if {
    $area_status eq "PASS" &&
    $wns_status eq "PASS" &&
    $tns_status eq "PASS"
} {

    set overall "PASS"

} else {

    set overall "FAIL"
}


# ========================================
# Display Summary
# ========================================

puts ""

vlsi::print_separator

puts "             QoR SUMMARY"

vlsi::print_separator

puts ""

puts "Cell Count       = $cell_count"
puts "NAND Count       = $nand_count"
puts "Total Area       = $total_area"

puts ""

puts "Largest Cell     = $largest_cell"
puts "Largest Area     = $largest_area"

puts ""

puts "WNS              = $wns"
puts "TNS              = $tns"
puts "Clock Period     = $period"

puts ""

puts "Area Status      = $area_status"
puts "WNS Status       = $wns_status"
puts "TNS Status       = $tns_status"

puts ""

puts "OVERALL STATUS   = $overall"

vlsi::print_separator
```

# 27. Example Real-Tool Style Logic

The exact command depends on the EDA tool, but the programming structure is:

```tcl
# Query design objects

set cells [EDA_TOOL_QUERY_CELLS]

# Process objects

foreach cell $cells {

    # Get property

    set area [EDA_TOOL_QUERY_AREA $cell]

    # Check property

    if {$area > 10} {

        puts "$cell has large area"
    }
}
```

The important part is not memorizing a particular command.

The important part is understanding:

```text
QUERY
 ↓

 ↓
FOREACH
 ↓
PROPERTY
 ↓

 ↓
```
